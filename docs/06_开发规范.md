## 2. 代码规范

### 2.1 命名约定

#### 文件命名
```
模块/功能.c
  例：core/engine.c
      hooks/buddy_hook.c
      builtin_strategies/numa_buddy.c

模块/功能.h
  例：smartmem.h
      strategy/strategy.h
```

#### 函数命名
```c
/* 模块_操作 */
smartmem_init(void)
smartmem_exit(void)

/* 对象_操作 */
buddy_strategy_register(struct buddy_strategy *s)
buddy_strategy_unregister(struct buddy_strategy *s)

/* 谓用/回调函数 */
buddy_pre_alloc(int order, gfp_t gfp_mask, int nid)
slub_post_alloc(void *object, size_t size)
```

#### 变量命名
```c
/* 全局变量：g_前缀 */
struct smartmem_state *g_sm_state;

/* 静态变量：s_前缀 */
static struct smartmem_state *s_sm_state;

/* 函数内：小写，描述性 */
int ret;
u64 alloc_count;
struct buddy_strategy *strategy;
```

#### 结构体命名
```c
/* 结构名：小写+下划线 */
struct smartmem_config {
    ...
};

struct buddy_strategy {
    ...
};

struct hotspot_info {
    ...
};
```

#### 宏命名

## 3. 编码规范

### 3.1 错误处理

#### 错误码定义
```c
/* 在smartmem.h中定义 */
#define SMARTMEM_SUCCESS        0
#define SMARTMEM_ERR_INVAL     1    /* 无效参数 */
#define SMARTMEM_ERR_NOMEM     2    /* 内存不足 */
#define SMARTMEM_ERR_EXIST     3    /* 已存在 */
#define SMARTMEM_ERR_NOTSUPP   4    /* 不支持 */
#define SMARTMEM_ERR_BUSY      5    /* 忙碌 */
```

#### 错误处理模式
```c
/* 1. 检查并返回 */
if (!ptr) {
    pr_err("smartmem: invalid pointer\n");
    return -EINVAL;
}

/* 2. 分配并检查 */
ptr = kzalloc(sizeof(*ptr), GFP_KERNEL);
if (!ptr) {
    pr_err("smartmem: allocation failed\n");
    return -ENOMEM;
}

/* 3. 错误传播 */
ret = function();
if (ret) {
    pr_warn("smartmem: function failed: %d\n", ret);
    return ret;
}
```

#### 资源清理
```c
int function(void)
{
    void *ptr1, *ptr2;

    ptr1 = alloc1();
    if (!ptr1)
        return -ENOMEM;

    ptr2 = alloc2();
    if (!ptr2) {
        free1(ptr1);
        return -ENOMEM;
    }

    /* ... */

    /* 成功 */
    free2(ptr2);
    free1(ptr1);
    return 0;
}
```

### 3.2 内存分配

#### 分配函数选择
```c
/* 核心数据结构：GFP_KERNEL */
g_state = kzalloc(sizeof(*g_state), GFP_KERNEL);

/* Hook路径：GFP_ATOMIC */
arg = kmalloc(sizeof(*arg), GFP_ATOMIC);

/* 策略注册：GFP_KERNEL */
strategy = kzalloc(sizeof(*strategy), GFP_KERNEL);
```

#### 分配大小计算
```c
/* 计算数组大小 */
size = count * sizeof(type);

/* 检查溢出 */
if (size / count != sizeof(type))
    return -EINVAL;

/* 分配 */
ptr = kzalloc(size, GFP_KERNEL);
```

### 3.3 并发控制

#### 锁选择
```c
/* 短持持锁时间短：spinlock */
spin_lock(&lock);
list_add(&item->list, &head);
spin_unlock(&lock);

/* 持持锁时间长：spinlock_irqsave */
unsigned long flags;
spin_lock_irqsave(&lock, flags);
do_something();
spin_unlock_irqrestore(&lock, flags);

/* 写多读少：rwlock */
read_lock(&rwlock);
do_read();
read_unlock(&rwlock);

write_lock(&rwlock);
do_write();
write_unlock(&rwlock);

/* 读多写少且需要升级：RCU */
rcu_read_lock();
do_read();
rcu_read_unlock();

synchronize_rcu();
do_write();
```

#### 死锁检查
```c
/* 确保获取锁顺序一致 */
lock1 = &s_state->lock;
lock2 = &strategy->lock;

mutex_lock(lock1);
mutex_lock(lock2);
...
mutex_unlock(lock2);
mutex_unlock(lock1);
```

### 3.4 统计更新

#### 原子操作
```c
/* 原子更新：无锁 */
atomic64_inc(&stats->count);
atomic64_add(&stats->bytes, size);

/* 原子读取 */
count = atomic64_read(&stats->count);
```

#### per-CPU统计
```c
/* per-CPU统计：无锁聚合 */
DEFINE_PER_CPU(struct cpu_stats, s_cpu_stats);

static inline void update_cpu_stat(void)
{
    struct cpu_stats *s = this_cpu_ptr(&s_cpu_stats);
    s->count++;
}

/* 聚合到全局 */
static void aggregate_cpu_stats(void)
{
    int cpu;
    for_each_possible_cpu(cpu) {
        struct cpu_stats *s = per_cpu_ptr(&s_cpu_stats, cpu);
        atomic64_add(&global_count, s->count);
    }
}
```

## 4. 性能规范

### 4.1 快速路径

```c
/* likely/unlikely优化 */
if (likely(strategy)) {
    /* 快速路径 */
}

if (unlikely(!ptr)) {
    /* 错误路径 */
}

/* 内联小函数 */
static inline void fast_path(void)
{
    /* 只有一两行代码 */
}
```

### 4.2 缓存局部性

```c
/* 热数据在栈上 */
struct temp_data data;
init_data(&data);
process(&data);

/* 重复使用放在栈上，不要堆分配 */
for (i = 0; i < 1000; i++) {
    struct work work;
    do_work(&work);
}
```

### 4.3 避免重复计算

```c
/* 缓存计算结果 */
static u64 cached_value;
static bool cached_valid;

static u64 expensive_calculation(void)
{
    if (cached_valid)
        return cached_value;

    cached_value = do_calculation();
    cached_valid = true;
    return cached_value;
}
```

### 4.4 批量操作

```c
/* 批量更新统计 */
for (i = 0; i < n; i++) {
    atomic64_inc(&stats->count);
}

/* 批量初始化 */
for (i = 0; i < n; i++) {
    items[i] = init_value;
}

/* 使用库函数 */
memset(array, 0, sizeof(array));
memcpy(dest, src, len);
```

### 4.5 零拷贝优化

```c
/* 避免不必要的拷贝 */
struct large_data *data = kmalloc(sizeof(*data), GFP_KERNEL);
if (!data)
    return;

/* 直接传递原指针 */
process(data);
kfree(data);

/* 而不是零拷贝 */
struct large_data *data = kmalloc(sizeof(*data), GFP_KERNEL);
struct large_data *copy = kmemdup(data, GFP_KERNEL);
process(copy);
kfree(data);
kfree(copy);
```

## 5. 安全规范

### 5.1 输入验证

```c
/* 指针验证 */
if (!ptr)
    return -EINVAL;

/* 大小验证 */
if (size > MAX_SIZE)
    return -EINVAL;

/* 范围验证 */
if (index < 0 || index >= count)
    return -EINVAL;

/* 字符串验证 */
if (len >= MAX_LEN)
    return -EINVAL;
```

### 5.2 边界检查

```c
/* 数组边界 */
for (i = 0; i < n; i++) {
    if (i >= n)
        break;
    process(&array[i]);
}

/* 链表边界 */
list_for_each_entry_safe(item, next, &head, list) {
    /* 处理时可能删除 */
    if (should_remove(item)) {
        list_del(&item->list);
        kfree(item);
    }
}
```

### 5.3 整数溢出

```c
/* 检查乘法溢出 */
if (count * size > MAX_SIZE)
    return -EINVAL;

/* 检查加法溢出 */
if (current + increment < current)
    return -EOVERFLOW;

/* 使用安全函数 */
size = array_size * count;
if (size / array_size != count)
    return -EINVAL;
```

### 5.4 NULL指针

```c
/* 使用前检查 */
if (ptr)
    use(ptr);

/* 解引用前检查 */
if (ptr && ptr->field)
    do_something(ptr->field);

/* 链表遍历 */
list_for_each_entry(item, &head, list) {
    if (item)  /* 检查 */
        process(item);
}
```

```
  - 确保无破坏
```

### 7.2 测试用例

```c
/* tests/test_buddy_strategy.c */

static int test_buddy_strategy_register(void)
{
    struct buddy_strategy *strategy;
    int ret;

    /* 创建测试策略 */
    strategy = kzalloc(sizeof(*strategy), GFP_KERNEL);
    if (!strategy)
        return -ENOMEM;

    snprintf(strategy->name, sizeof(strategy->name), "test");
    strategy->ops.pre_alloc = test_pre_alloc;

    /* 测试注册 */
    ret = buddy_strategy_register(strategy);
    if (ret) {
        pr_err("test: register failed: %d\n", ret);
        kfree(strategy);
        return ret;
    }

    /* 测试查询 */
    struct buddy_strategy *s = buddy_strategy_get();
    if (s != strategy) {
        pr_err("test: get failed\n");
        buddy_strategy_unregister(strategy);
        kfree(strategy);
        return -EINVAL;
    }

    /* 测试注销 */
    buddy_strategy_unregister(strategy);
    kfree(strategy);

    return 0;
}
```
